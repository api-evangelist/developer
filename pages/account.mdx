export const SignInButtons = ({ base }) => {
  const [providers, setProviders] = useState(null);
  useEffect(() => {
    // Ask the API which providers are actually configured rather than hard-coding three buttons —
    // a button for a provider whose credentials are not deployed is a dead end for the user.
    fetch(`${base}/auth/providers`)
      .then((r) => (r.ok ? r.json() : null))
      .then((j) => setProviders(j?.providers || []))
      .catch(() => setProviders([{ id: 'github', label: 'GitHub', login: `${base}/auth/login?provider=github` }]));
  }, [base]);
  if (providers === null) return <p>Loading sign-in options…</p>;
  if (!providers.length) return <p>Sign-in is temporarily unavailable.</p>;
  return (
    <div>
      <div style={{ display: 'flex', gap: '0.6rem', flexWrap: 'wrap', margin: '0.5rem 0 0.75rem' }}>
        {providers.map((p) => (
          <a key={p.id} href={p.login} style={{
            display: 'inline-flex', alignItems: 'center', gap: '0.55rem',
            padding: '0.55rem 0.95rem', borderRadius: '8px',
            border: '1px solid var(--border, #d0d7de)', textDecoration: 'none',
            fontWeight: 600, fontSize: '0.92rem', lineHeight: 1,
            color: 'inherit', background: 'var(--background, transparent)',
          }}>
            {MARKS[p.id]}<span>Sign in with {p.label}</span>
          </a>
        ))}
      </div>
      <p style={{ fontSize: '0.85rem', color: '#666', marginTop: 0 }}>
        Signing in with a second provider on the <strong>same verified email address</strong> joins
        the account you already have — your key and plan follow you, rather than a duplicate account
        being created.
      </p>
    </div>
  );
};

export const AE_BASE = 'https://api.apievangelist.com/v1';

export const Account = () => {
  const [me, setMe] = useState(undefined);
  useEffect(() => {
    fetch(`${AE_BASE}/auth/me`, { credentials: 'include' })
      .then((r) => (r.ok ? r.json() : null))
      .then(setMe)
      .catch(() => setMe(null));
  }, []);
  if (me === undefined) return <p>Loading…</p>;
  if (me === null) return (
    <div>
      <p>Sign in to create your account and a free-tier API key — no email required.</p>
      <SignInButtons base={AE_BASE} />
    </div>
  );
  return (
    <div>
      <p>Signed in as <strong>{me.login}</strong> — current tier: <strong>{me.tier}</strong>
        {me.identities?.length ? <> · via {me.identities.join(', ')}</> : null}</p>
      <p><strong>Your API key</strong> (send as the <code>x-api-key</code> header):</p>
      <pre>{me.apiKey}</pre>
      <pre>{`curl "${me.baseUrl}/guide?topic=api%20governance" -H "x-api-key: ${me.apiKey}"`}</pre>
      <p style={{ fontSize: '0.85rem' }}>
        MCP server: <code>{me.mcp}</code> · <a href={`${AE_BASE}/auth/logout`}>Log out</a>
      </p>
    </div>
  );
};

This is where sign-in and Stripe checkout return you. Your session lives in a
secure cookie on `apievangelist.com`; your personal API key is minted the first time you
sign in.

## Get your key

<Account />

The page reads `/v1/auth/me` with your session cookie, so your key appears above once you are
signed in — the same JSON is available at
[`/v1/auth/me`](https://api.apievangelist.com/v1/auth/me) if you would rather read it directly.

Send the key as `x-api-key` on any Pro endpoint:

   ```bash
   curl -X POST https://api.apievangelist.com/v1/governance/toolsmith \
     -H "x-api-key: ae_live_…" -H "content-type: application/json" \
     -d '{"document":"openapi: 3.0.0\n..."}'
   ```

## Upgrade to Pro

Start a subscription — your existing key is moved onto the Pro (or Team) usage plan
automatically when checkout completes:

```bash
curl -X POST https://api.apievangelist.com/v1/billing/checkout \
  --cookie "ae_session=…" -H "content-type: application/json" \
  -d '{"tier":"pro","cycle":"monthly"}'
# → { "url": "https://checkout.stripe.com/..." }
```

Manage or cancel any time via the billing portal:

```bash
curl -X POST https://api.apievangelist.com/v1/billing/portal --cookie "ae_session=…"
```

See [the plans](/plans) for what Pro unlocks.
